# tm_gpu: CPU references are always compiled; at most one GPU backend is selected.
add_library(tm_gpu STATIC philox_fill_cpu.cpp field_cpu.cpp)
add_library(tm::gpu ALIAS tm_gpu)
target_link_libraries(tm_gpu PUBLIC tm::core)
tm_set_warnings(tm_gpu)

if(TM_ENABLE_CUDA)
  target_sources(tm_gpu PRIVATE device_cuda.cu philox_fill_cuda.cu field_cuda.cu)
  target_link_libraries(tm_gpu PRIVATE CUDA::cudart)
  target_compile_definitions(tm_gpu PUBLIC TM_HAVE_CUDA=1 TM_HAVE_METAL=0)
  set_target_properties(tm_gpu PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
elseif(TM_ENABLE_METAL)
  target_sources(tm_gpu PRIVATE device_metal.mm field_metal.mm)
  target_compile_options(tm_gpu PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  find_library(TM_METAL_FRAMEWORK Metal REQUIRED)
  find_library(TM_FOUNDATION_FRAMEWORK Foundation REQUIRED)
  target_link_libraries(tm_gpu PRIVATE ${TM_METAL_FRAMEWORK} ${TM_FOUNDATION_FRAMEWORK})
  target_compile_definitions(tm_gpu PUBLIC TM_HAVE_CUDA=0 TM_HAVE_METAL=1)
else()
  target_sources(tm_gpu PRIVATE device_stub.cpp)
  target_compile_definitions(tm_gpu PUBLIC TM_HAVE_CUDA=0 TM_HAVE_METAL=0)
endif()
