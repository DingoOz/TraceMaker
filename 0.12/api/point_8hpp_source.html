20struct Box {
21 Coord x0 = 0, y0 = 0, x1 = -1, y1 = -1; // empty when x1 < x0
22 constexpr bool empty() const { return x1 < x0 || y1 < y0; }
23 constexpr void add(Point p) {
24 if (empty()) { x0 = x1 = p.x; y0 = y1 = p.y; return; }
25 if (p.x < x0) x0 = p.x;
26 if (p.x > x1) x1 = p.x;
27 if (p.y < y0) y0 = p.y;
28 if (p.y > y1) y1 = p.y;
29 }
30 constexpr void add(const Box& b) {
31 if (b.empty()) return;
32 add(Point{b.x0, b.y0});
33 add(Point{b.x1, b.y1});
34 }
35 constexpr Box inflated(Coord d) const { return empty() ? *this : Box{x0 - d, y0 - d, x1 + d, y1 + d}; }
36 constexpr bool intersects(const Box& b) const { return !(b.x0 > x1 || b.x1 < x0 || b.y0 > y1 || b.y1 < y0); }
37};

42// Normalises degrees to [0, 360).
43inline double norm_deg(double a) {

45 if (a < 0) a += 360.0;

50// Rotates p by `deg` degrees the way KiCad does: positive angles turn counter-clockwise on screen in KiCad's
51// y-down coordinates. Multiples of 90° are exact.
52inline Point rotate(Point p, double deg) {
53 const double a = norm_deg(deg);
54 if (a == 0.0) return p;
55 if (a == 90.0) return {p.y, -p.x};
56 if (a == 180.0) return {-p.x, -p.y};
57 if (a == 270.0) return {-p.y, p.x};
58 const double r = a * std::numbers::pi / 180.0;
59 const double c = std::cos(r), s = std::sin(r);
60 const double x = static_cast<double>(p.x), y = static_cast<double>(p.y);
61 return {kiround(x * c + y * s), kiround(-x * s + y * c)};
62}
