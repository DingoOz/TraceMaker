14struct CopperItem {
15 ItemKind kind = ItemKind::Track;
16 int index = -1; // index in the board's pads/tracks/arcs/vias/zones/graphics vector
17 int sub = 0; // zone fill polygon index
18 model::NetId net = 0;
19 model::LayerMask layers = 0; // copper layers the shape is on
20 std::vector<geom::Shape> shapes; // copper shape (identical on every layer in `layers`)
21 geom::Box box;
22 int footprint = -1;
23 model::Point pos; // reporting position
24 Coord width = 0; // tracks/arcs: width; vias: diameter
25 int owner = -1; // router connection that created this item (-1 = fixed copper)
26 bool removed = false; // ripped up (router working model only)
27 bool free_via = false; // via marked (free yes): its net is fixed for KiCad's net propagation
28 // Vias and pads whose copper KiCad removes on layers nothing connects to (model::UnconnectedLayers, doc 05 §60). On those
29 // layers only the hole is left, and KiCad's clearance test takes the hole for the item's copper there. Such an item is split:
30 // the item with the copper (`layers` without the bare layers) and a second item of the same kind and index whose layers are
31 // the bare ones and whose shape is the hole. `bare_part` on the first points to the second, `bare_of` on the second to the
32 // first. The second is no node of the connectivity graph and is not reported on its own. `span` is the padstack's whole layer
33 // set (KiCad's GetLayerSet), which rule conditions see; 0 for items that are not split.
34 model::LayerMask span = 0;
35 int bare_part = -1, bare_of = -1;
36};
