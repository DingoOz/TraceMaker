3// Internal: the incremental annealing state shared by simulated annealing, parallel tempering, large-
4// neighbourhood search and exact windows (anneal.cpp, tempering.cpp, lns.cpp). Not part of the public API.

21inline constexpr Coord kLattice = 10'000; // committed positions on a 10 µm lattice
22inline constexpr int kMaxMove = 12; // parts in one compound move (LNS windows)
23inline constexpr double kLnsPhase = 0.3; // LNS windows run in the last 30 % of an annealing budget
24
25inline Coord snap(Coord v) { return (v >= 0 ? (v + kLattice / 2) : (v - kLattice / 2)) / kLattice * kLattice; }
26inline Point snap(Point q) { return Point{snap(q.x), snap(q.y)}; }
27inline std::int64_t alpha_units(double alpha_mm) { return static_cast<std::int64_t>(std::llround(alpha_mm * 1e6)) * kSignalWeight; }
28// Cost of one nm of RUDY overflow: β (mm of signal HPWL per mm of overflow) × signal weight; resolution 0.1.
29// Cost of one estimated via per unit of net weight: a via on a signal net (weight kSignalWeight) costs via_mm of
30// signal HPWL.
31inline std::int64_t via_units(double via_mm) { return static_cast<std::int64_t>(std::llround(via_mm * 1e6)); }
32inline std::int64_t beta_units(double beta) { return static_cast<std::int64_t>(std::llround(beta * kSignalWeight)); }
33
34inline Point body_centre(const Part& pt, int rot) {
35 const Box& b = pt.geom[z(rot)].body;
36 return Point{(b.x0 + b.x1) / 2, (b.y0 + b.y1) / 2};
37}
