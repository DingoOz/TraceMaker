3// Lossless s-expression documents for KiCad files (design doc 08 §3).
4//
5// The parser keeps the original text and records each node's byte span, so writing an unmodified
6// document reproduces the input byte for byte. Edits are recorded as span replacements and insertions
7// and applied on write; untouched text is never re-formatted.

9#include <cstdint>

19using NodeId = std::uint32_t;
20inline constexpr NodeId kNoNode = 0xFFFFFFFFu;
21
22enum class Kind : std::uint8_t { List, Symbol, String };
23
24struct Node {
25 Kind kind;
26 std::uint32_t begin; // byte offset of '(' / first char of the atom / opening quote
27 std::uint32_t end; // one past ')' / last char / closing quote
28 std::uint32_t first_child; // index into Document::kids_ (lists only)
29 std::uint32_t child_count;
30 NodeId parent;
31};

33class ParseError : public std::runtime_error {
34 public:
35 ParseError(const std::string& msg, std::size_t offset, int line)
36 : std::runtime_error(msg + " at line " + std::to_string(line)), offset_(offset), line_(line) {}
37 std::size_t offset() const { return offset_; }
38 int line() const { return line_; }
39
40 private:
41 std::size_t offset_;
42 int line_;
43};
