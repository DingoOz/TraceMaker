13// The length a via adds to its net in KiCad's length calculation (KiCad 10, BOARD_STACKUP::GetLayerDistance): the
14// stackup between the two copper layers the via joins, with the copper of an outer layer in full and of an inner
15// layer by half. A board without a stackup block gets KiCad's default one (35 um copper, 10 um per solder mask, the
16// rest of the board thickness shared evenly between the dielectrics). Empty (every via 0) when the project turns
17// "use height for length calculations" off.
18struct ViaHeights {
19 std::vector<Coord> from, to; // per copper layer: where a via starting / ending on it begins / ends, nm from the top
20 Coord between(int a, int b) const; // copper indices, either order; 0 for a == b
21};
22ViaHeights via_heights(const model::Board& b, const model::DesignRules& rules);

24// The items of one net that a length or skew rule matched: indices into the board's pads, tracks, arcs and vias.

26 std::vector<int> pads, tracks, arcs, vias;

29// KiCad's LENGTH_DELAY_STATS. Lengths in nm, delays in attoseconds.
30struct NetLength {
31 std::int64_t track = 0; // tracks and arcs, joined into chains and clipped in vias and pads
32 Coord via = 0; // via heights (0 when the project does not use heights)
33 Coord die = 0; // pad-to-die lengths
34 int vias = 0, pads = 0;
35 std::int64_t track_delay = 0, via_delay = 0, die_delay = 0;
36 double total() const { return static_cast<double>(track) + static_cast<double>(via) + static_cast<double>(die); }
37 double total_delay() const {
38 return static_cast<double>(track_delay) + static_cast<double>(via_delay) + static_cast<double>(die_delay);
39 }
40};
41
42// Length and delay of `items`, all of one net. `profile` is the tuning profile of the net's class (null: every
43// delay is 0, as in KiCad).
44NetLength net_length(const model::Board& b, const model::DesignRules& rules, const ViaHeights& vh, const LengthItems& items,
45 const model::TuningProfile* profile);
