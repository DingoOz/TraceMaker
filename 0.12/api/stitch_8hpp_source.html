3// Stitching vias (doc 05 §39.3): after routing, vias of a plane's net are added where fills of that net lie over
4// each other on two or more layers (plane stitching), and next to the vias of signals that need a reference plane
5// (return-path stitching). Every via passes the router's exact check against the refilled zones, and is kept only
6// if a second refill still joins it to its net's fill on two layers.
7#include <string>
8#include <vector>
9
10#include "model/board.hpp"
11#include "model/rules.hpp"

13namespace tmk::passes {
14
15struct StitchOptions {
16 bool planes = false; // plane stitching on a grid
17 // Grid pitch. The default is the catalogue's general high-speed value (TI SPRAAR7J: a stitching via within
18 // 200 mil); app code passes the catalogue's number, this is its fallback.
19 Coord pitch = 5'080'000;
20 // Return-path stitching: every via of `signal` wants a via (or plated through-hole pad) of `plane` within
21 // `distance`, centre to centre.
22 struct ReturnNet {
23 model::NetId signal = 0, plane = 0;
24 Coord distance = 5'080'000;
25 std::string source; // "option" or the component rule (e.g. "USB2-09 J1")
26 };
27 std::vector<ReturnNet> returns;
28 // Work budget (rule 5): sites examined plus exact checks. When it runs out no further site is tried.
29 long work_limit = 2'000'000;
30};
31
32struct StitchReturn {
33 model::NetId signal = 0, plane = 0;
34 std::string source;
35 Coord distance = 0;
36 int signal_vias = 0; // vias of the signal net on the board (old and new)
37 int stitched_before = 0; // of them: with a plane via or plated pad within the distance before the pass
38 int stitched_after = 0; // ... and after it
39};
40
41struct StitchResult {
42 std::vector<model::Via> vias; // the vias to add, plane stitches first; all marked free
43 std::vector<std::uint8_t> return_path; // per via: 1 = placed for a signal via, 0 = grid
44 std::vector<StitchReturn> returns; // one row per StitchOptions::returns entry
45 int sites = 0; // grid sites inside overlapping fills
46 int site_served = 0; // of them: already tied by a via or plated pad of the net within half the pitch
47 int site_blocked = 0; // of them: refused by the exact check
48 int dropped = 0; // vias placed and then removed because the second refill did not join them on two layers
49 long checks = 0; // the pass's work
50 bool out_of_work = false;
51 bool discarded = false; // everything withdrawn: the refill with the vias left another net's zone in more pieces
52 std::vector<std::string> warnings;
53};
