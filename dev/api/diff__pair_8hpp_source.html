18// What a pair is routed at. Sources, highest first: a custom rule's `diff_pair_gap` (opt, else min) and
19// `diff_pair_uncoupled` (max); the net class's diff-pair width/gap/via gap when the project sets them; else the net
20// class width and the clearance between the two halves (the tightest legal coupling). Never below the board minimums
21// or the clearance KiCad requires between the halves (the class diff-pair gap relaxes it only for pairs KiCad
22// recognises by name, `inDiffPair`).
23struct PairRule {
24 Coord width = 0; // track width of each half
25 Coord gap = 0; // edge-to-edge gap of the coupled section (target; >= required)
26 Coord required_gap = 0; // smallest gap the rules allow between the halves
27 Coord offset = 0; // centreline to each track centre: (width + gap) / 2, rounded up, plus a rounding margin
28 Coord via_diameter = 0, via_drill = 0;
29 Coord via_offset = 0; // centreline to each via centre in a coupled via pair (>= offset)
30 std::optional<Coord> max_uncoupled; // per half, both ends together
31 std::string source; // "rule", "net class" or "clearance" (for the report)
32};

48// A routed pair, measured on its tracks: a sample of one half counts as coupled where a parallel track of the
49// other half on the same layer lies within `coupled_gap` edge to edge.
50struct PairStats {
51 std::string net_a, net_b;
52 double length_a = 0, length_b = 0; // track length, nm
53 double coupled_a = 0, coupled_b = 0; // coupled track length, nm
54 int vias_a = 0, vias_b = 0;
55 double gap_min = 0, gap_median = 0; // edge gap over coupled samples, nm (0 when nothing is coupled)
56 double coupled_share() const { return length_a + length_b > 0 ? (coupled_a + coupled_b) / (length_a + length_b) : 0.0; }
57 double skew() const { return length_a > length_b ? length_a - length_b : length_b - length_a; }
58};
59// Edge gap up to which a sample counts as coupled: the pair's gap plus a quarter of it (at least 50 um), so lattice
60// rounding and the miter at bends do not count as uncoupled while a single track wandering off does.
61inline Coord coupled_threshold(const PairRule& r) { return r.gap + std::max<Coord>(r.gap / 4, 50'000); }
62PairStats measure_pair(const std::vector<model::Track>& tracks, const std::vector<model::Via>& vias, model::NetId a, model::NetId c,
63 Coord coupled_gap);
