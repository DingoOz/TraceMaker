3// The advice file (design doc 17, issue #26): typed hints for the router, written by a person, a script or a language
4// model, and read by the engine like any other input. The engine never talks to a model; a model is one possible writer
5// of this file.
6//
7// {
8// "version": 1,
9// "note": "free text, logged and never acted on",
10// "variants": 4, // advised variants added to the portfolio (default 4)
11// "first_nets": ["USB_DP", "USB_DM"], // routed first; other nets do not rip them
12// "priority": [["U1.A1", "J2.3"]], // connections (pad to pad, either order) routed first
13// "layers": {"no_tracks": ["In2.Cu"], // no new tracks on these copper layers (vias still pass)
14// "cost": {"F.Cu": 1.5}}, // track cost factor per layer (1 to 1000)
15// "regions": [{"name": "north channel", // a region of the board (millimetres, the board file's axes)
16// "rect": [10, 10, 20, 14], // or "polygon": [[x, y], ...]
17// "layers": ["F.Cu"], // default: every copper layer
18// "reserve_for": ["NET1", "NET2"], // only these nets may put copper there ...
19// "avoid_nets": ["NET3"], // ... or: these nets may not (exactly one of the two)
20// "tracks": true, "vias": true}] // which copper is kept out (both by default)
21// }
22//
23// Advice never contains copper and never changes a rule's value: it changes the order of the connections, the cost of
24// a layer, and where some nets may not go, all inside the router's search. Every route is still judged by the exact check,
25// and the advised attempts run as extra portfolio variants next to the unadvised ones, so the winner can only be better.
26//
27// Reading is strict, as for the rule overrides (doc 15 §6.3): an unknown key, a wrong type or a value out of range is an
28// error naming the file and the entry, never ignored (a typo must not silently drop a hint). Names are checked against the
29// board by check_advice_refs; entries that have no effect are warnings.

56struct Region {
57 std::string name; // "" = numbered
58 std::vector<std::string> layers; // copper layer names; empty = every copper layer
59 std::vector<geom::Point> polygon; // nm, the board file's axes
60 bool tracks = true, vias = true;
61 std::vector<std::string> reserve_for; // only these nets may enter
62 std::vector<std::string> avoid_nets; // these nets may not enter
63 std::string where; // "regions[2]" in messages
64};

66struct Advice {
67 std::string source; // file name, for messages
68 int version = 1;
69 std::string note;
70 int variants = kDefaultVariants;
71 std::vector<std::string> first_nets;
72 std::vector<std::pair<std::string, std::string>> priority; // "REF.NUM" pairs
73 std::vector<std::string> no_tracks; // copper layer names
74 std::vector<std::pair<std::string, double>> layer_cost;
75 std::vector<Region> regions;
76 bool empty() const { return first_nets.empty() && priority.empty() && no_tracks.empty() && layer_cost.empty() && regions.empty(); }
77};
