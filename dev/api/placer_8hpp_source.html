4// C (rotations) → D (legalisation) → E (annealing); `refine` starts from the current placement and runs
5// D (only for illegal parts) → E. Both report the exact lower bounds of doc 04 §1 (L4) and what was achieved.

19struct PlaceOptions {
20 std::string mode = "full"; // full | refine
21 std::uint64_t seed = 1;
22 int threads = 0; // 0 = min(hardware, 16)
23 int runs = 0; // annealing runs (0 = threads)
24 double effort = 4.0; // annealing moves per run = effort × 4000 × movable parts
25 double alpha_cross_mm = 2.0;
26 double time_limit_s = 0; // wall-time stop for the annealer (0 = none)
27 bool verbose = false;
28 // M8 (doc 04 §3 E, G): see AnnealOptions.
29 bool tempering = false; // parallel tempering over the runs instead of independent runs
30 double lns_rate = 0; // share of annealing steps that are LNS windows
31 int lns_window = 6;
32 int lns_polish = 0; // LNS windows run on the annealing result (deterministic polish; 0 = none)
33 double beta_congestion = 0; // routability weight (0 = off)
34 const CongestionMap* congestion = nullptr;
35 std::vector<int> focus; // LNS seed parts
36 // Side assignment (doc 04 §3 C/E, D48): see AnnealOptions. The problem must be extracted with
37 // ExtractOptions::flip for any part to be allowed to flip.
38 bool flip = false;
39 double via_mm = 2.0;
40 double flip_rate = 0.1;
41 // Recording (video): receives intermediate placements with a stage name. Never affects the result.
42 // `t` is when the state existed (trace_now()); annealing snapshots are delivered after the run.
43 std::function<void(const std::string& stage, const Placement& pl, double t)> trace;
44};

46struct Metrics {
47 std::int64_t whpwl = 0; // Σ weight·HPWL (signal nets weight 10, power nets 1), nm
48 std::int64_t hpwl = 0; // Σ HPWL over all nets, nm
49 std::int64_t crossings = 0; // MST airwire crossings between different signal nets
50 int flipped = 0; // parts on the other side than in the input
51 std::int64_t side_vias = 0; // via estimate of the side assignment (unweighted side_vias)
52 int overlaps = 0, outside = 0; // violations involving movable parts
53 int fixed_overlaps = 0, fixed_outside = 0;
54 // Relative to a reference (the input) placement: violations where at least one involved movable part was
55 // moved. Pairs whose parts both stayed put existed in the input and are not ours.
56 int new_overlaps = 0, new_outside = 0;
57 std::vector<std::string> conflicts; // "A/B" pairs and "A/outline" (movable-involved)
58};

72 double legalise_mean_disp_mm = 0, legalise_max_disp_mm = 0;
73 std::int64_t legal_start_whpwl = 0; // weighted HPWL entering the annealer
74 AnnealResult anneal;
75 std::uint64_t lns_tried = 0, lns_improved = 0; // polish stage
76 double seconds_total = 0;
77 std::vector<std::pair<std::string, double>> stage_seconds;
