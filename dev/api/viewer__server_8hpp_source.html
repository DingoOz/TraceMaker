3// HTTP + WebSocket server for the live viewer (design doc 09, protocol in docs/13-viewer-protocol.md).
4//
5// Serves the built viewer (static files) at http://<host>:<port>/ and streams protocol messages at /ws.
6// publish() is thread-safe and never blocks on the network: messages are queued and fanned out by a single
7// background I/O thread. The server keeps the latest `board` snapshot plus the state changes since it, so a
8// client that connects mid-run sees the current scene. Slow clients lose transient messages (frontier,
9// path_try) and get coalesced stats; a client that falls hopelessly behind is disconnected (it reconnects and
10// receives a fresh snapshot).
11#include <cstddef>
12#include <cstdint>
13#include <memory>
14#include <string>

16#include "core/events.hpp"

20// Who can reach the viewer (issue #4, docs/09-visualization.md "Security", D116): by default only this machine. A viewer bound to any other address
21// requires a token on the page and on /ws (a random one unless `token` is given; `allow_open` serves without, a deliberate choice that is said so),
22// and every WebSocket upgrade must come from the server's own origin, so a web page open in the user's browser cannot read the stream.
23// The bind address when none is given: $TM_VIEW_HOST if set, else 127.0.0.1 (this machine only).
24std::string default_host();
25
26struct ServerOptions {
27 std::string host = default_host(); // "0.0.0.0" or a LAN address makes the viewer reachable from the network: it then needs a token
28 std::uint16_t port = 8766; // 0 = pick a free port (tests)
29 std::string token; // at least 16 characters of [A-Za-z0-9._~-]; empty = generated when the host is not a loopback address
30 bool allow_open = false; // a non-loopback host without any token (--view-insecure)
31 std::string embed; // pages that may frame the viewer, as CSP sources separated by spaces (--view-embed); empty = nobody (X-Frame-Options: DENY)
32 std::size_t max_clients = 16; // WebSocket viewers, counted from the upgrade on (a burst cannot overshoot); the next gets 503
33 std::size_t max_connections = 128; // concurrent HTTP connections (a page load is one); beyond it a connection is closed at once
34 std::size_t max_connections_per_address = 16; // of those, from one address: one host cannot use them all up before it has authenticated
35 std::string web_root; // directory with index.html; empty = <source dir>/viewer/dist
36 std::size_t max_queue_msgs = 2000; // per client: beyond this, transient messages are dropped
37 std::size_t max_queue_bytes = 8u << 20; // per client: same, by bytes
38 std::size_t disconnect_bytes = 256u << 20; // per client: state backlog that drops the connection
39};
