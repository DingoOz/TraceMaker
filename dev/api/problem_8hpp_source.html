50// Orientation state of a part (Placement::rot): bits 0–1 = quarter turns r, bit 2 = flipped to the other side.
51// State r means angle0 + 90 r on the input side; state 4 + r means the KiCad flip (top/bottom about the origin:
52// offsets mirrored in y, orientation −angle0) followed by r quarter turns, i.e. −angle0 + 90 r on the other side.
53inline constexpr int kStates = 8;
54inline constexpr std::uint8_t kFlipBit = 4;
55inline bool flipped(int state) { return (state & kFlipBit) != 0; }
56// State with the same side as `state` and `r` quarter turns.
57inline std::uint8_t with_turn(int state, int r) { return static_cast<std::uint8_t>((state & kFlipBit) | (r & 3)); }
58inline Point mirror_y(Point p) { return Point{p.x, -p.y}; }
59// Exchanging two interchangeable parts a and b (same footprint and input side; angle0_b − angle0_a = 90° dk): the
60// state that gives a the absolute pose b has in `b_state`. Flipping negates angle0, so dk changes sign there.
61// (b takes a's pose with swap_state(a_state, −dk).)
62inline std::uint8_t swap_state(int b_state, int dk) { return with_turn(b_state, (b_state & 3) + (flipped(b_state) ? -dk : dk) + 4); }

64struct Part {
65 int fp = -1; // footprint index in the board
66 std::string ref, lib_id;
67 bool movable = false;
68 bool copper_only = false; // ExtractOptions::copper_only: only its copper and holes are tested against other parts
69 std::string fixed_reason; // why a part is fixed (locked, mounting hole, ...)
70 int side = 0; // 0 front, 1 back: the footprint's side in the input
71 // May move to the other side (states 4–7): movable, surface mount only (no holes), the board has two copper
72 // sides, flipping was asked for, and the writer can mirror the footprint (doc 04 §3 C/E, D48).
73 bool flippable = false;
74 std::string flip_reason; // why a movable part may not flip (empty when it may, or when flipping is off)
75 Point pos0; // original origin
76 double angle0 = 0; // original absolute orientation (degrees); see kStates for state angles
77 std::array<PartGeom, kStates> geom; // per state (states 4–7 only filled for flippable parts)
78 bool may_flip() const { return movable && flippable; } // movability can be revoked after extraction
79 int side_in(int state) const { return side ^ (flipped(state) ? 1 : 0); }
80 double angle_of(int state) const { return flipped(state) ? -angle0 + 90.0 * (state & 3) : angle0 + 90.0 * (state & 3); }
81 std::vector<int> pins; // indices into Problem::pins
82 Coord area = 0; // courtyard box area incl. clearance (nm², saturating), for spreading
83 std::uint64_t shape_key = 0; // equal keys = interchangeable footprints (swap moves)
84};

141// An objective-only pull of a part toward the board outline (doc 15 CONN-01, opt-in): the body point nearest to the
142// outline in the input becomes a one-pin pseudo-net anchored on the nearest outline segment's axis (x for a mostly
143// vertical segment, y for a mostly horizontal one), so the pull is exact in the annealer's integer cost and
144// rotating the part away from the edge costs too.
145struct EdgePull {
146 int footprint = -1;
147 std::vector<Point> body; // absolute candidate points (courtyard outline, else pad corners)
148 int weight = 2 * kSignalWeight;
149 std::string name; // "~CONN-01 J3 edge"
150};

152struct ExtractOptions {
153 bool fix_edge_connectors = true; // connectors (J*, P*, CN*, USB*) touching the outline stay put
154 Coord courtyard_clearance = -1; // override (-1: from the rules, else default_clearance)
155 Coord default_clearance = 250'000; // used when the board has no courtyard rule (KiCad's own default is 0)
156 bool decap_affinity = true; // tie each decoupling capacitor to the nearest supply pin of its IC
157 int decap_weight = kSignalWeight; // weight of those ties (D25: signal weight)
158 int crules_weight_pct = 100; // scale of the component-rule proximity weights (100 = as the rules say)
159 // Full mode with component rules: place with the decoupling ties only, then lock the tied capacitors and refine
160 // with the component-rule pulls (they compete for the space next to the IC otherwise; doc 15 §14.3).
161 bool crules_two_stage = true;
162 // Extra objective-only pseudo-nets from component rules (crules::placement_affinities). Parts already tied by
163 // decap_affinity are skipped.
164 std::vector<PadAffinity> affinities;
165 // Edge pulls for movable parts (crules::edge_attractions); fixed parts are skipped.
166 std::vector<EdgePull> edge_pulls;
167 // Side assignment (doc 04 §3 C/E, D48): movable surface-mount parts may move to the other side. `flip_ok` (one
168 // entry per footprint; empty = all) says whether the writer can mirror the footprint, `flip_why` why not;
169 // `keep_side` lists references the user pins to their side.
170 bool flip = false;
171 std::vector<std::uint8_t> flip_ok;
172 std::vector<std::string> flip_why;
173 std::vector<std::string> keep_side;
174 // References whose courtyard is ignored: the part is as large as its pads (box + 0.25 mm), like a footprint
175 // drawn without a courtyard. Full mode's fallback for parts whose courtyard fits nowhere (designers overlap
176 // courtyards routinely: 24 of the 40 boards of set H do; doc 04 §9).
177 std::vector<std::string> pads_only;
178 // References tested by copper and holes only: neither their outline nor the other parts' courtyards count
179 // against them (copper clearance, hole spacing and the board edge still do). The last resort before a part is
180 // left off the board: battery holders, modules and headers that the designer placed across other courtyards.
181 std::vector<std::string> copper_only;
182 // The input positions of movable parts mean nothing (a board straight from the schematic: parts piled or beside
183 // the board). A part that overhangs the edge where the input has it is then not held there.
184 bool scratch = false;
185};
