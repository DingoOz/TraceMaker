20struct Role {
21 std::string name;
22 std::vector<int> parts; // footprint indices
23 std::vector<int> pads; // pad indices (pins playing the role, e.g. an IC's XTAL pins)
24 std::vector<model::NetId> nets;
25 bool empty() const { return parts.empty() && pads.empty() && nets.empty(); }
26};

28// What the user override file (doc 15 §6.3) changed for one rule of one instance.

37struct Instance {
38 int category = -1; // index into Catalogue::categories
39 int anchor = -1; // footprint index
40 int confidence = 0; // 0..100
41 bool capped = false; // confidence capped because the board has no pin names (doc 15 §3.6)
42 std::vector<std::string> evidence; // matched detectors, e.g. "lib_id +60"
43 std::vector<Role> roles; // bound roles, binding order
44 std::string superseded_by; // set when a conflicting category won the same anchor (doc 15 §3.4)
45 bool asserted = false; // forced by the user override file (`assert`)
46 std::vector<RuleOverride> overrides; // from `disable` and `set` entries, one per rule id
47 const Role* role(std::string_view name) const;
48 const RuleOverride* override_for(std::string_view rule) const;

50};

69// With `ov` (the user override file, doc 15 §3.5 level 1): references are checked against the board (throws on
70// unknown ones); `deny` removes a category from a part (listed as possible, so a competing category can win);
71// `assert` forces one (confidence 100, wins conflicts; bound by the category's binder, or to the anchor alone when
72// the binder rejects the part); `disable` and `set` are attached to the instances: entries for one reference
73// (@REF) take precedence over global ones, and among equals the later entry wins.
