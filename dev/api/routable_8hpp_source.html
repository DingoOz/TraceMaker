3// Router-in-the-loop placement (design doc 04 §3 G) and ECO placement (doc 04 §4).
4//
5// Both take the router as a callback (`RouteFn`), so this library does not depend on tm::route and the logic
6// can be tested with a synthetic router. The caller routes with a deterministic work budget, so every decision
7// here is reproducible. A placement is only ever accepted when it routes more connections than the incumbent
8// (transactions, CLAUDE.md rule 4); locked and otherwise fixed parts never move (rule 6).

20struct RouteEval {
21 bool ok = false;
22 int connections = 0, routed = 0;
23 int unrouted() const { return connections - routed; }
24 struct Failure {
25 int part_a = -1, part_b = -1; // parts at both ends (-1: a zone or an unknown reference)
26 Point a, b; // end points (pad positions; b = a for zone connections)

28 };
29 std::vector<Failure> failed;

31};
32using RouteFn = std::function<RouteEval(const Placement&)>;

75// Small moves of movable parts near the failed connections of `start` (shift, rotate, swap with an identical
76// part), ranked by a geometric estimate (annealing cost with boosted failed nets, penalised congestion bins and a
77// displacement charge), the top ones routed; the best strictly improving move is committed per round. Moves that
78// failed are remembered and not tried again.
79EcoResult eco_place(const Problem& p, const Placement& start, const RouteEval& start_eval, const EcoOptions& o, const RouteFn& route);

83struct LoopOptions {
84 PlaceOptions place; // annealing options for the re-placement rounds (mode is forced to refine)
85 double beta = 1.0; // routability weight (RUDY overflow) in the re-placement rounds
86 int rounds = 3; // penalty rounds after the seeds
87 double penalty = 0.6; // capacity factor applied per round to bins around failed connections
88 int weight_boost = 3; // failed nets weigh this many times more in the re-placement rounds
89 int eco_candidates = 4; // ECO moves routed per round (0 = no ECO step)
90 std::function<void(const std::string&)> log;
91 // Job wall-time stop (CLAUDE.md rule 5): checked between routes; when it returns true the loop stops and keeps
92 // the best candidate so far. The first seed (the input) is always routed.
93 std::function<bool()> out_of_time;
94 // Recording (video): called with a candidate's label whenever it becomes the incumbent. Re-placement stages
95 // reach o.place.trace as "<label>|<stage>".
96 std::function<void(const std::string& label)> on_incumbent;
97 std::function<void(const std::string& label)> on_route; // recording: a check route of this candidate starts
98 // --repair-input (D75): only seeds without any conflict can become the incumbent; the others are routed for
99 // reference. Off: every seed is eligible, as its caller checked it adds no conflict to the input's.
100 bool absolute_seed_legality = false;
101 // With absolute_seed_legality: the first seed is the input whose minimal repair failed; it is compared as it is.
102 bool accept_first_seed_conflicts = false;
103};
104
105struct LoopResult {
106 Candidate best;
107 std::vector<Candidate> tried; // every routed candidate (placements dropped), in order
108 int routes = 0, rounds = 0;
109};

111// Routes the seeds (those without a result), keeps the best, then for up to `rounds` rounds re-places around the
112// failed connections (refine annealing with β·RUDY over a capacity map shrunk where the router failed, boosted
113// weights for the failed nets, LNS windows centred on their parts) and tries ECO moves; a candidate replaces
114// the incumbent only if it leaves fewer connections unrouted.
115LoopResult routability_loop(const Problem& p, std::vector<Candidate> seeds, const LoopOptions& o, const RouteFn& route);
