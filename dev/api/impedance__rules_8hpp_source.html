3// Impedance and current rules on a real board (design doc 15 §5.3-5.4, P4): the board stackup gives, per routing
4// layer, the line structure and the dielectric to the nearest reference layer; the closed forms of impedance.hpp
5// turn a rule's target (z0_ohm / zdiff_ohm) into a width and gap. Report-only: nothing here changes routing.
6//
7// Assumption (dev/assumptions-m13.md, P4): the copper layer adjacent to a routing layer is its reference plane.
8// Outer layers are microstrip over the next copper layer; inner layers are stripline between the copper layers
9// above and below. A zone on the reference layer is reported as confirmation; without one the report says the
10// plane is assumed. Boards without a stackup get nothing computed (doc 15 §3.6; CLAUDE.md rule 6).

24// One copper layer's line geometry from the stackup.
25struct LayerGeometry {
26 int copper = -1;
27 std::string layer;
28 bool routing = true; // false for KiCad "power" (plane) layers
29 imp::Structure structure = imp::Structure::Microstrip; // outer: microstrip; inner: stripline
30 int ref_a = -1, ref_b = -1; // reference copper layers (microstrip: ref_a only; stripline: above, below)
31 std::string ref_a_note, ref_b_note; // "zone GND" or "no zone: assumed plane"
32 Coord h1 = 0, h2 = 0; // dielectric to ref_a (microstrip) / to the planes above and below (stripline)
33 Coord t = 0; // copper thickness of this layer
34 double er = 0; // dielectric permittivity (stripline: series combination of both sides)
35 bool ok = false;
36 std::string why; // when !ok
37};
38std::vector<LayerGeometry> stackup_geometry(const model::Board& b);

40// The width (and gap) that meets one target on one layer.
41struct LineSolution {
42 std::string layer, ref; // "F.Cu", "In1.Cu (zone GND)"
43 imp::Structure structure = imp::Structure::Microstrip;
44 bool differential = false;
45 double target = 0; // ohm
46 bool ok = false;
47 std::string why; // when !ok
48 Coord width = 0, gap = 0; // gap: pair gap (differential) or coplanar ground gap (GCPW); 0 otherwise
49 double z = 0; // achieved impedance at the integer-nm solution
50 double eps_eff = 0, ps_per_mm = 0; // differential: odd mode
51 double error_pct = 0; // stated formula error
52 Coord h = 0, h2 = 0; // dielectric to the reference (stripline: above, below)
53 double er = 0;
54 std::string note; // e.g. "gap raised: width at the board minimum"
55};

57struct ImpedancePlan {
