18 std::string in; // input .kicad_pcb (its project files supply the design rules)
19 std::string out; // output .kicad_pcb; empty = do not write a board
20 route::RouterOptions opt; // `sink` is set here when viewing or recording
21 int threads = 8; // worker threads for the portfolio variants
22 // Portfolio size (1 = single router). 0 = automatic: all variants (route::portfolio_size()) with a work budget,
23 // so the output does not depend on `threads` (D47); otherwise `threads` (wall-clock mode: one variant per thread).
24 int variants = 0;
25 std::string kb_path; // knowledge base file; empty = none
26 std::string items_out; // --emit-items file; empty = none
27 std::string json_out; // --json summary file; empty = none
28 bool view = false; // stream to the browser viewer
29 // Reference-return audit of selected nets on the routed board (issue #7, doc 05 §42): off unless a selector is given. Read-only; the
30 // report goes to the log and the summary, and each net's return path to the viewer.
31 analysis::ReturnPathOptions return_paths;
32 // Critical nets (issue #24, doc 05 §44, D117): route critical signals first, give them dearer vias, keep them over solid reference copper. Off
33 // unless asked; `si_rules` turns the three on with the settings below unless they were given.
34 std::vector<std::string> critical_nets; // KiCad wildcards: nets that are critical whatever their role (tier 1)
35 bool critical_auto = false; // also the nets the component rules' roles make critical, and differential pairs
36 bool critical_first = false; // route them before all other nets, tier by tier
37 double critical_via_mult = 1; // a via costs them this many times as much
38 bool critical_via_given = false; // set by the command line: with si_rules, a value that was given is kept (else x3)
39 int critical_via_max = 0; // warn when a critical net has more vias than this (0 = no limit)
40 double split_cost = 0; // pitches per lattice step over a void of the reference plane
41 bool split_cost_given = false; // as critical_via_given (else +4 with si_rules)
42 bool split_hard = false; // such cells are closed to critical nets
43 bool si_rules = false;
44 // --footprint-refs (issue #5, doc 05 §43): every footprint is checked against the datasheet land patterns of a reference library before the first track is
45 // routed; a discrepancy is an error and stops the job unless footprint_warn_only.
46 std::string footprint_refs; // the library: a directory or a JSON file; empty = no check
47 std::string footprint_trusted; // trusted brand list; empty = trusted_brands.txt beside the references
48 bool footprint_trust_all = false;
49 bool footprint_warn_only = false;
50 analysis::FootprintCheckOptions footprint_opt;
51 bool view_freeze = true; // the viewer holds the scene on a routing stall (issue #8); --no-freeze says false in the board snapshot
52 std::string view_host; // "" = $TM_VIEW_HOST, else loopback: a viewer on any other address needs a token (issue #4, D116)
53 std::string view_token; // the token for a non-loopback viewer (empty = a random one)
54 bool view_insecure = false; // a non-loopback viewer without a token (--view-insecure)
55 int view_port = 8766;
56 bool hold = false; // keep serving the viewer after routing; never returns (CLI only)
57 std::string record; // events file for replay (ignored while viewing)
58 // --component-rules (doc 15 §3.5): "off" (default), "report" or "soft" (detect; write the sidecar .kicad_dru next
59 // to the output), "on" (also route with the generated keep-outs as in-memory rule areas; never written into the
60 // output board).
61 std::string component_rules = "off";
62 // --reroute: remove every unlocked track and via first and route the whole board again (locked copper, arcs and
63 // zones stay; rule 6).
64 bool reroute = false;
65 std::string warm_pcb; // --warm-pcb: a routed board whose tracks and vias are the starting point (kept where legal under the input's rules; nets of one connection)
66 std::string rules_override; // --rules-override: user override file (JSON, doc 15 §6.3); empty = none
67 // Advice (doc 17, issue #26). The engine never talks to a model: it reads an advice file, and an advisor (any program
68 // that turns a stuck report into an advice file) is consulted only when the deterministic route leaves connections open.
69 std::string advice; // --advice FILE: typed hints (src/advice/advice.hpp); advised variants run next to the unadvised ones
70 int advice_variants = 0; // --advice-variants N: advised variants (0 = the file's "variants", else 4)
71 std::string advisor; // --advisor CMD: a shell command that reads the stuck report on stdin and writes advice on stdout
72 double advisor_timeout_s = 120; // --advisor-timeout S: the advisor's time limit (its wall time is not routing work)
73 int advisor_rounds = 1; // --advisor-rounds N: consult again with the result of the advised run while connections are left
74 bool advisor_redact = false; // --advisor-redact: the report names nets and parts by alias (N1, P3); the advice is mapped back
75 std::string advice_out; // --advice-out FILE: where the advice that was used is saved (default: <out>.advice.json)
76 std::string stuck_report_out; // --stuck-report FILE: write the stuck report (of the run the advisor saw, else the final one)
77 // With soft zones: refill the zones after routing and route what the refill left unconnected (doc 05 §36).
78 bool refill_repair = true;
79 // --no-tracks-on / --layer-cost (doc 05 §29): copper layer names, and "LAYER=FACTOR" entries. Resolved against
80 // the board into opt.no_track_layers / opt.layer_cost_pm; an unknown layer or a bad factor throws.
81 std::vector<std::string> no_tracks_on;
82 std::vector<std::string> layer_costs;
83 // Buses (doc 05 §38): "NAME=PATTERN" groups (--bus), automatic detection (--buses) and the skew limit of the
84 // command-line groups in mm (--bus-skew-mm; 0 = none). A custom skew rule of the board wins over the option.
85 std::vector<std::string> buses;
86 bool detect_buses = false;
87 double bus_skew_mm = 0;
88 // After-route passes (doc 05 §39), all off by default. Each changes the new copper only through the router's
89 // exact check and is kept only if no net has more unconnected items with it than without (rule 4).
90 // --widen-power: new tracks narrower than their net class width go back up to it where there is room; nets with
91 // a target (--widen-net "PATTERN=WIDTH_MM", or a width the component rules computed) are widened towards it.
92 bool widen_power = false;
93 std::vector<std::string> widen_nets;
94 // --stitch-vias: "planes" (vias of a plane's net on a grid where its fills overlap on two or more layers),
95 // "return" (a plane via near each via of the signals that need a reference plane: component rules of kind
96 // reference_plane, and --stitch-return "PATTERN=PLANE_NET"), or "all". Empty = off.
97 std::string stitch_vias;
98 double stitch_pitch_mm = 0; // 0 = the catalogue's stitch_via_within_mm
99 double stitch_distance_mm = 0; // 0 = the rule's stitch_via_mm, else the catalogue value
100 std::vector<std::string> stitch_returns;
101 long pass_work = 2'000'000; // work budget of each pass, in exact checks (rule 5)
102 // Progress lines (the CLI's stdout text, one line per call, no trailing newline). Empty = silent.
103 std::function<void(const std::string&)> log;
104};

106struct RouteJobResult {
107 route::RouteResult result;
108 nlohmann::json items; // new copper in the --emit-items layout (layer and net by name, nm)
109 nlohmann::json summary; // the --json layout
110 std::string viewer_url; // empty when not viewing
111 int unconnected_after_refill = -1; // with soft zones: what KiCad's refill would leave unconnected
112 // 0 when everything planned was routed (and, with soft zones, nothing is unconnected after the refill).
113 int exit_code() const { return unconnected_after_refill >= 0 ? (unconnected_after_refill == 0 ? 0 : 3) : result.routed == result.connections ? 0 : 3; }
114};

127// Lengths are KiCad's: tracks plus the stackup height of the vias (`rules` says whether the project counts it).
