19 void add_track(const model::Track& t);

23 // Moves footprint `index` to `pos` with orientation `angle` (degrees). Pad and text orientations, which
24 // KiCad stores as absolute angles, are rotated by the same delta. Flipping sides is not supported here.
25 void move_footprint(std::size_t index, model::Point pos, double angle);
26 // Moves footprint `index` to the other side of the board the way KiCad does (FOOTPRINT::Flip, top/bottom, about
27 // its own origin: local y and pad orientations negated, text angles 180° − a, F.* <-> B.* layers, texts
28 // mirrored), then puts it at `pos` with the final orientation `angle` in degrees (flipped orientation −angle0
29 // plus any rotation). Only for footprints where flip_supported() holds.
30 void flip_footprint(std::size_t index, model::Point pos, double angle);
