2#pragma once
3// Footprint check against datasheet land patterns (issue #5, docs/05-routing.md §43, D115).
4//
5// For each footprint of a board the check finds the reference (footprint_ref.hpp) made for it: by manufacturer part number, library id, value or
6// designator. A reference from a brand that is not on the trusted list gives no verdict (the footprint is reported as `untrusted`: nobody vouched for
7// that datasheet). With a trusted reference the footprint's pads are compared with the datasheet's: the pattern is first aligned to the footprint
8// (a rotation by a multiple of 90 degrees, a mirror, and a translation: the footprint's origin and orientation are the library's choice, not
9// a property of the land pattern), then every pad is checked for number, position, size, drill and type. Whatever differs is an error.
10//
11// A mirror is not free: the pads of a footprint on the front must match the datasheet as drawn, those of a footprint on the back mirrored (that is
12// what flipping a footprint does). A pattern that fits only the other way round is reported as mirrored: the pin numbers run the wrong way.
13//
14// The check is read-only and geometric. It verifies the footprint against the record somebody made from the datasheet; it does not read the datasheet.

26enum class Severity { Info, Warning, Error };

28struct FootprintIssue {
29 Severity severity = Severity::Error;
30 std::string kind; // pad_count, missing_pad, extra_pad, position, size, drill, pad_type, orientation, mirrored, pattern, no_reference, untrusted_source
31 std::string pad; // the pad number, "" = the footprint as a whole
32 std::string message;
33 std::string label; // a few words for a mark on the board ("pad 3: 0.15 mm off")
34 geom::Point at; // where to look on the board (nm)
35 double actual_mm = -1; // the measured value where there is one (the deviation of a position, a size ...)
36 double expected_mm = -1;
37};
38
39enum class FootprintStatus { Ok, Mismatch, Unverified, Untrusted, Skipped };
40
41struct FootprintResult {
42 int footprint = -1;
43 std::string ref, lib_id, value;
44 std::vector<std::string> mpn;
45 geom::Point at;
46 FootprintStatus status = FootprintStatus::Unverified;
47 // The reference that was used (Ok, Mismatch) or found but not trusted (Untrusted).
48 std::string ref_id, brand, part, document, revision, figure, url;
49 int page = 0;
50 int other_matches = 0; // further references that also matched and were not used
51 int rotation = 0; // the alignment found: degrees the datasheet pattern is turned to meet the footprint
52 bool mirrored = false;
53 int pads_board = 0, pads_ref = 0, pads_matched = 0;
54 double max_position_mm = 0; // the largest deviation of a matched pad after alignment
55 std::vector<FootprintIssue> issues;
56 int errors() const;
57 int warnings() const;
58};
59
60struct FootprintCheckOptions {
61 double position_tol_mm = 0.05; // a reference may set its own
62 double size_tol_mm = 0.05;
63 double drill_tol_mm = 0.05;
64 std::vector<std::string> only; // designator patterns to check (empty = all)
65 std::vector<std::string> require; // designator patterns that must have a trusted reference: without one it is an error
66 std::vector<std::string> mpn_fields; // property names (besides the usual MPN ones) that hold the manufacturer part number
67};
68
69struct FootprintCheckReport {
70 std::vector<FootprintResult> results; // board order
71 int checked = 0, ok = 0, mismatched = 0, unverified = 0, untrusted = 0, skipped = 0;
72 int errors = 0, warnings = 0;
73 std::vector<std::string> notes; // what the run could not do or chose to do (trust everything, ambiguous matches ...)
74 FootprintCheckOptions options;
75 int references = 0;
76 std::string trusted_source; // the trusted list's file, "" = none
77 bool trust_all = false;
78};
