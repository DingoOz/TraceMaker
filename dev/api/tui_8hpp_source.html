2// `tracemaker tui`: choose the options of a command on a terminal screen (doc 02 §2.1, D89).
3//
4// The screen is a list of the command's options, taken from the command-line definitions themselves, so it cannot
5// fall behind them. `Model` holds the state and is free of terminal code: keys go in, lines of text come out. The
6// terminal layer below it only reads keys and paints those lines.
7#pragma once

10#include <string>
11#include <vector>

15// Where a value comes from, lowest precedence first. `run` is what belongs to one run (board, output) and is never
16// written to a file; `global` is the user's options for every board, `project` the options of the board's folder and
17// `file` a file named with --config. A higher scope overrides a lower one.
18enum class Scope { run = 0, global = 1, project = 2, file = 3 };
19inline constexpr int kScopes = 4;
20const char* scope_name(Scope s); // "this run", "global", "project", "file"
21
22// One command-line option as the screen shows it.
23struct Field {
24 std::string label; // as shown: "--via-cost-mm", "-o, --output", "board"
25 std::string arg; // spelling written before the value; empty for a positional argument
26 std::string key; // options-file key of a value option: its long name without the dashes
27 std::vector<std::string> spellings; // a flag's spellings ("--pair-twists", "--no-pair-twists"); empty for a value option
28 std::string help;
29 std::string def; // the default as text; empty if not known
30 bool hidden = false; // experimental and test options, shown on request
31 bool save = true; // false for what belongs to one run (board, output): not written to an options file
32 std::string value; // what is in effect: value option: its text; flag: the chosen spelling; empty: the default
33 std::string layer[kScopes]; // the same, per scope; `value` is the highest scope's that is not empty
34 Scope scope = Scope::run; // the scope `value` comes from (meaningless while value is empty)
35 bool is_flag() const { return !spellings.empty(); }
36};
37
38// Keys beyond the printable characters, which stand for themselves.
39namespace key {
40inline constexpr int ctrl_c = 3, tab = 9, enter = 13, ctrl_u = 21, esc = 27, backspace = 127;
41inline constexpr int up = 0x1000, down = 0x1001, page_up = 0x1002, page_down = 0x1003, home = 0x1004, end = 0x1005, del = 0x1006, shift_tab = 0x1007;
42}

44// One line of an options file: `name = value`. Leading dashes on the name, quotes around the value and a TOML
45// array ([a, b] becomes a,b) are accepted; a name alone means `name = true`. `#` and `;` start a comment line.
46struct Entry {
47 int line = 0;
48 std::string name, value;
49};
50std::vector<Entry> parse_options(const std::string& text);
51// A flag's value in an options file: 1 for true/on/yes/1, 0 for false/off/no/0, -1 for anything else.
52int truth(const std::string& value);
53
54enum class Action { none, quit, run, print, save };
55
56// How a piece of text is painted. The terminal layer maps these to colours (or to bold, dim and reverse under
57// NO_COLOR); the model only says what the text is.
58enum class Style {
59 plain, frame, title, heading, dim, label, key, code, ok, warn, error, accent,
60 global, project, file, run, // a value or a name of that scope
61 tag_global, tag_project, tag_file, tag_run // the same as a badge
62};
63struct Span {
64 std::string text;
65 Style style = Style::plain;
66};
67// One screen line: `text` is the plain text (what a test or a monochrome terminal sees), `spans` the same in pieces
68// with their styles. `highlight` paints the whole line as the current row.
69struct Line {
70 std::string text;
71 std::vector<Span> spans;
72 bool highlight = false;
73};
74// Columns the text takes (UTF-8 characters count one each, which is right for the box-drawing characters).
75std::size_t display_width(const std::string& s);
