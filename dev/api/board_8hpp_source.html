47struct Pad {
48 int footprint = -1;
49 std::string number;
50 PadType type = PadType::Smd;
51 PadShape shape = PadShape::Rect;
52 Point pos; // absolute centre
53 double angle = 0; // absolute orientation, degrees
54 Coord size_x = 0, size_y = 0;
55 Coord drill_x = 0, drill_y = 0; // 0 = no hole
56 bool drill_oval = false;
57 Point drill_offset; // copper shape offset from the hole (pad position), in the pad's own frame
58 LayerMask copper = 0; // copper layers the pad has copper on
59 std::vector<std::string> layers; // all layer names as written (incl. mask/paste, wildcards expanded)
60 double roundrect_ratio = 0;
61 double chamfer_ratio = 0;
62 std::uint8_t chamfer_corners = 0; // bit 0 TL, 1 TR, 2 BL, 3 BR
63 Coord trapezoid_dx = 0, trapezoid_dy = 0; // rect_delta
64 // Custom-pad primitives in the pad frame, each already grown by half its stroke width (KiCad draws them with a round pen and fills
65 // them): a polygon or rectangle, a polyline (line, arc) or a disc (circle; pts[0] is the centre).
66 struct CustomPrim {
67 enum class Kind { Polygon, Polyline, Disc } kind = Kind::Polygon;
68 std::vector<Point> pts;
69 Coord radius = 0;
70 };
71 std::vector<CustomPrim> custom_prims;
72 Coord clearance = -1; // local override, -1 = none
73 Coord mask_margin = INT64_MIN; // local solder-mask expansion (unset = INT64_MIN)
74 ZoneConnect zone_connect = ZoneConnect::Inherited; // (zone_connect n)
75 Coord thermal_gap = -1, thermal_bridge_width = -1; // local overrides, -1 = none
76 double thermal_bridge_angle = -1; // degrees; -1 = KiCad's default (45 for round pads, 90 otherwise)
77 // Schematic pin name and electrical type (KiCad 6+ boards; empty in older files). Read-only, used to
78 // recognise component roles (doc 15 §3.1).
79 std::string pinfunction, pintype;
80 NetId net = 0;
81 sexpr::NodeId node = sexpr::kNoNode;
82};
