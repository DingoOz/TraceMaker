3// Builders for viewer protocol v1 messages (docs/13-viewer-protocol.md). Pure functions returning one JSON
4// object per message, ready for events::Sink::publish(). No Boost or networking: the router links only this
5// library (tm::viz) and publishes into whatever Sink it was given.
6//
7// Every message starts with its "type" key (insertion-ordered JSON), so a server can classify messages by
8// looking at the first bytes without parsing them.

54struct Failure {
55 std::int64_t conn = 0;
56 model::NetId net = 0;
57 int rung = 0;
58 std::string cause;
59 model::Point a, b;
60 std::vector<ObjectId> blockers;
61 geom::Box region; // empty box = none
62 // Why the attempt was rejected (doc 13, issue #8); all optional, and an old reader ignores them. Lengths in nm; unset = left out.
63 std::vector<model::NetId> blocker_nets; // nets of the connections the rejected route would have ripped
64 std::string reason; // boxed_in, no_legal_path, exact_check, rip_limit, ...
65 std::int64_t width_nm = 0, clearance_nm = 0, min_width_nm = 0; // the attempt's track width and clearance, and the narrowest legal width
66 std::int64_t gap_nm = -1, gap_fixed_nm = -1; // widest track that gets through (routed copper counted / only fixed copper)
67 int gap_layer = -1;
68 std::optional<model::Point> at; // where the way is narrowest (or the closest approach)
69 std::string verdict; // placement | rule | contested | unknown
70 std::string rule; // where the width and clearance came from
71 std::vector<PathPoint> rejected; // the path that was tried
72};
