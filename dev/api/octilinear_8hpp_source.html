3// Octilinear clean-up (doc 05 §80, D187, M23 E3): a new track that runs off the eight directions (0, 45, 90, ... degrees) is replaced by
4// a dogleg of two tracks that are on them, through one of the two points that split it into a diagonal and a straight leg. A dogleg is
5// kept only if both legs are free in the router's exact check, against everything including the copper already changed, and if neither
6// end makes an acute corner with the copper it joins. Ends do not move, so connectivity is unchanged.
7#include <vector>
8
9#include "model/board.hpp"
10#include "model/rules.hpp"
11
12namespace tmk::passes {
13
14struct OctilinearOptions {
15 long work_limit = 2'000'000; // exact checks (rule 5); when it runs out the remaining tracks stay as they are
16 bool soft_zones = false; // as RouterOptions::soft_zones
17 // A leg shorter than this many track widths would be a jog of its own (the style metric's short segment): such a dogleg is not
18 // used. 0 allows any length.
19 double min_leg_widths = 0.5;
20};
21
22struct OctilinearResult {
23 int off_angle = 0; // new, unlocked tracks off the eight directions before the pass
24 int replaced = 0; // of them: replaced by a dogleg
25 int no_room = 0; // a dogleg that fits nowhere (both candidates failed the exact check)
26 int short_leg = 0; // skipped: a leg would be shorter than min_leg_widths
27 int acute = 0; // skipped: a dogleg would make an acute corner with its neighbours
28 long checks = 0;
29 bool out_of_work = false;
30};

32// `board` is the board as the router saw it (without the new copper); `tracks` and `vias` are the run's new copper. On return
33// `tracks` holds the doglegs in place of the tracks they replace (each replaced track becomes two, in its place in the order).
34OctilinearResult octilinear_tracks(const model::Board& board, const model::DesignRules& rules, std::vector<model::Track>& tracks,
