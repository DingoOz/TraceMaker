3// Live status of this process for the progress site's live page (devsite /live, D165): what the process is doing, written as
4// JSON to <dir>/<pid>.json about once a second by a background thread (written to a temporary file and renamed, so a reader
5// never sees half a file). <dir> is $TM_LIVE_DIR, else $XDG_CACHE_HOME/tracemaker/live, else ~/.cache/tracemaker/live.
6// On by default; TM_LIVE=0 turns it off. A process that ends within its first second writes nothing.
7//
8// Write-only: the engine never reads any of it back, so it cannot change a result (rule 2). The router updates its slot with
9// relaxed atomic stores, which cost next to nothing next to a search; everything else takes a mutex a few times a second.
10#include <atomic>
11#include <string>
12
13namespace tmk::live {

15// One router's progress (a portfolio variant, by name), updated lock-free from the thread running it.
16struct Slot {
17 std::atomic<int> routed{0}, total{0};
18 std::atomic<long> work{0}, budget{0}; // search expansions of the run in progress and its work budget (0: none)
19 std::atomic<long> work_done{0}; // search expansions of the slot's finished runs
20 std::atomic<int> phase{0}; // 0 waiting, 1 setup, 2 first pass, 3 negotiation, 4 clean-up, 5 done
21 std::atomic<int> runs{0}; // router runs started in this slot (one per round)
22};

27void finish(int exit_code);
