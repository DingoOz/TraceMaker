3// From detected instances to effective rules and engine inputs (design doc 15 §3.5, §5.5, §7):
4// evaluate -> every catalogue rule of every instance, tagged applied / satisfied-by-board / not-applied(reason)

6// compile -> placement pseudo-nets (pad pairs), keep-out rule areas for the router and DRC, and a sidecar
7// .kicad_dru with the generated custom rules;
8// report -> the text and JSON the user sees.

56// A proximity rule bound to pads (doc 15 §4 `proximity`): each pair is (pad of the part that should move close,

70// Placement pseudo-nets for every applicable proximity rule (mode soft or on; confidence >= suggest). A part is

79// Connector edge pull for the placer (doc 15 §5.2 `edge`; CONN-01, USB2-11, DISP-02), opt-in
80// (tracemaker-place --edge-attraction): the anchor of every instance with a connector edge rule (mode soft/on, not
81// advisory, not disabled by the user), once per footprint, never a locked part and never a pin header or socket
82// (board-to-board and jumper headers, CONN-01 note). `body` holds the courtyard outline points (else pad corners),
83// absolute; the placer pulls the body point nearest to the board outline toward the outline.
84struct EdgeAttraction {
85 int footprint = -1;
86 std::vector<Point> body;
87 int weight = 20; // 2 x signal weight, like ESD-01 (doc 15 §5.2)
88 double max_mm = 0; // the rule's limit (report only)
89 std::string rule, name; // "CONN-01", "~CONN-01 J3 edge"
90};
91std::vector<EdgeAttraction> edge_attractions(const model::Board& b, const Catalogue& cat, const Detection& det, Mode mode);
92
93// Keep-out rule areas (doc 15 §5.5, P2) for instances at confidence >= apply: crystal + load caps (XTAL-04),
94// switching-regulator inductor (BUCK-06/BOOST-04), discrete Ethernet magnetics (ETH-05: tracks, vias and zones on
95// the magnetics' side and the adjacent layer only). The polygon is the convex hull of the parts' courtyards (and the
96// IC's oscillator pins for a crystal) grown by the rule margin, on the copper layers where none of those parts has
97// a pad, so the parts' own nets still reach their pads; tracks only (vias and zones stay allowed: ground stitching
98// under a crystal is good practice). Zones are named "tmk:<rule>:<ref>".
99struct GeneratedKeepout {
100 model::Zone zone;
101 std::string rule, ref;
102 int instance = -1;
103};
104std::vector<GeneratedKeepout> generate_keepouts(const model::Board& b, const Catalogue& cat, const Detection& det,
105 std::vector<std::string>* not_generated = nullptr);
106// The same intent as KiCad custom rules with net exemptions (which rule areas cannot express), for the user to
107// review or merge: "<output>.tracemaker.kicad_dru". Never written into the user's own files (rule 8).
108std::string dru_sidecar(const model::Board& b, const Catalogue& cat, const Detection& det);

110// USB 2.0 D+/D- pairs to route coupled first (USB2-02, doc 15 P3): one (dp, dm) per usb2 instance whose dp and dm
111// roles each bind exactly one net, in instance order, without duplicates.

114// Net classes from rules (doc 15 §5.3-5.4 and §16, P5): impedance rules with a computed width/gap, and width rules
115// with a current or a minimum width, become synthetic net classes for the nets bound to the rule's roles. Only for
116// instances at the apply threshold, only for nets the board leaves in its Default class (the board's own classes
117// always win, §3.5), and only ever wider than the class the net has (never looser than a board rule). The router
118// has one width per net: the outer-layer solution is used and `detail` says when inner layers would differ.
119struct SyntheticClass {
120 model::NetClass cls; // name "tmk_<RULE>_<anchor reference>"
121 std::vector<model::NetId> nets;
122 std::string rule, detail;
123};
124std::vector<SyntheticClass> synthetic_net_classes(const model::Board& b, const model::DesignRules& rules, const Catalogue& cat,
125 const Detection& det, const Evaluation& ev);
126// `rules` with the synthetic classes added and their nets assigned to them (in memory only, rule 8).
127model::DesignRules with_synthetic_classes(const model::Board& b, model::DesignRules rules, const std::vector<SyntheticClass>& sc);
