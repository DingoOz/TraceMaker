2// `tracemaker tui` jobs screen (doc 02 §2.2, D193): the route runs of this user, read from the live status files every
3// run writes (core/live.hpp), with what each has left to do, the options it was given, and what to change on a rerun.
4//
5// Like the options screen, `JobsModel` is free of terminal code: jobs and keys go in, lines of text come out. The
6// advice is a pure function of one job (`advise`), so it is tested on recorded jobs.

22// One router of a job (a portfolio variant), as its live file shows it.
23struct JobSlot {
24 std::string name;
25 int routed = 0, total = 0, phase = 0, runs = 0;
26};
27
28// One `route` process, from its live file and, once it has finished, the summary it wrote with --json.
29struct Job {
30 std::string file; // the live file
31 int pid = 0;
32 std::string state; // running, done (exit code known), ended (left without one), gone (the file says running, the process is not there)
33 int exit_code = -1;
34 double started = 0, updated = 0, elapsed_s = 0, cpu_s = 0;
35 std::vector<std::string> argv; // the arguments after `tracemaker route`
36 std::string cwd;
37 std::vector<std::pair<std::string, std::string>> fields; // stage, round, result ... in the order the run set them
38 std::vector<JobSlot> slots;
39 std::vector<int> h_routed, h_total; // samples every h_step_s seconds
40 int h_step_s = 1;
41 std::vector<std::string> log;
42 nlohmann::json summary; // null until the run has written it
43
44 std::string field(const std::string& key) const; // "" if not set
45 std::optional<std::string> option(const std::string& spelling) const; // a value option's value; nullopt if not given
46 bool has(const std::string& spelling) const; // a flag or option given
47 std::string path(const std::string& p) const; // p against the job's folder
48 std::string board() const; // the input board as given
49 int routed() const; // best so far (the summary's once finished)
50 int total() const; // connections, 0 while unknown
51 double budget_s() const; // the wall-time limit read from the "budget" field, 0 if none
52 double stalled_s() const; // seconds since the best routed count last rose (from the history)
53};

62// What to try next for a job, with the data it rests on. `set` are the options a rerun would take (a flag has an
63// empty value), `unset` those it would drop. Advice with neither is information only.
64struct Advice {
65 std::string text, why;
66 std::vector<std::pair<std::string, std::string>> set;
67 std::vector<std::string> unset;
68};
69std::vector<Advice> advise(const Job& j);

84 void set_status(std::string s) { status_ = std::move(s); }
