3// Component-rule catalogue (design doc 15 §6.1): categories of components with their detectors, roles and
4// layout rules. The source is docs/component_rules.yaml; scripts/crules_catalogue.py converts it to
5// src/crules/catalogue.json, which the build embeds (the engine has no YAML parser), and a ctest keeps the
6// two identical.

18// What a detector looks at (doc 15 §3.1).
19enum class Signal : std::uint8_t { RefPrefix, LibId, Value, Keywords, PinName, NetName, Topology };
20enum class Severity : std::uint8_t { Hard, Soft, Advisory };
21
22const char* signal_name(Signal s);
23const char* severity_name(Severity s);
24
25struct Detector {
26 Signal signal = Signal::RefPrefix;
27 std::string pattern; // ECMAScript regex (case-insensitive), or a topology name for Signal::Topology
28 int weight = 0; // 0..100; confidence = min(100, sum of matching weights)
29 std::regex re; // compiled `pattern` (unused for topology)
30};
31
32struct RuleSpec {
33 std::string id, kind, text, evidence, note;
34 std::vector<std::string> applies_to, enforce, sources;
35 nlohmann::ordered_json params;
36 Severity severity = Severity::Soft;
37 bool enforced_in(std::string_view stage) const;
38};
39
40struct Category {
41 std::string id, name, prefix;
42 std::vector<Detector> detect;
43 std::vector<std::pair<std::string, std::string>> roles; // name -> description, catalogue order
44 std::vector<RuleSpec> rules;
45 bool has_signal(Signal s) const;
46};
47
48struct Catalogue {
49 int version = 0;
50 int apply = 70, suggest = 40; // confidence thresholds (doc 15 §3.2)
51 // Skew ps -> mm without a stackup (doc 15 §4, §5.3): conservative FR-4 delays, outer / inner layers.
52 double prop_delay_outer = 6.0, prop_delay_inner = 7.0;
53 // General high-speed guidance (defaults.general_high_speed): a stitching via within this distance of a layer change.
54 double stitch_via_within_mm = 5.08;
55 std::vector<Category> categories;
56 int index_of(std::string_view id) const; // -1 if absent
57 std::size_t rule_count() const;
58};

63Catalogue load_catalogue_file(const std::string& path);
64// The catalogue embedded at build time (src/crules/catalogue.json), parsed once.
65const Catalogue& builtin_catalogue();
66std::string_view builtin_catalogue_json();
