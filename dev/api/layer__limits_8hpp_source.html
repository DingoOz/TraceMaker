13struct LayerLimits {

18inline constexpr double kMaxLayerCost = 1000.0;
19
20// `no_tracks_on`: copper layer names. `layer_costs`: "NAME=FACTOR" entries with 1 <= FACTOR <= kMaxLayerCost.
21// Names are the board's canonical names ("In1.Cu"), the names written in the file, or the user's layer names.
22// Throws std::invalid_argument on an unknown layer, a malformed entry, a factor out of range, or when no layer
23// would be left for tracks: a limit the router cannot apply as written must stop the run, not be dropped.
24LayerLimits layer_limits(const model::Board& b, const std::vector<std::string>& no_tracks_on, const std::vector<std::string>& layer_costs);

26// Preferred track direction of a copper layer (issue #66, D174): orthogonal steps across it cost --layer-dir-pen.
27enum LayerDir : std::int8_t { kLayerDirNone = 0, kLayerDirHorizontal = 1, kLayerDirVertical = 2 };
28
29// Per-layer directions and diagonal penalties (`--layer-dir`, `--layer-diag-pen`; doc 05 §29, D174) resolved against a board.
30struct LayerDirections {
31 std::vector<std::int8_t> dir; // per copper index (LayerDir); empty = not given (the layer parity of D98 applies)
32 std::vector<double> diag_pen; // per copper index: pitches per 45-degree step; empty = not given
33 std::vector<std::string> warnings; // entries naming a layer the board does not have (ignored)
34};
35
36inline constexpr double kMaxDiagPen = 100.0;
37
38// Syntax check without a board, for the command line: `dir_specs` are "LAYER=h|v|none" entries, `diag_specs` are
39// "LAYER=P" entries or one bare "P" for every copper layer (0 <= P <= kMaxDiagPen). Throws std::invalid_argument on a
40// malformed entry, a bad value or a layer given twice.
41void check_layer_dir_specs(const std::vector<std::string>& dir_specs, const std::vector<std::string>& diag_specs);
42
43// The entries resolved against `b` (names as for layer_limits). A layer name the board does not have is a warning and
44// the entry is ignored (a board-independent option file may name inner layers); everything check_layer_dir_specs refuses
45// still throws. With dir_specs given, layers not named have no preferred direction; with diag_specs given, layers not
46// named (and no bare value) have no diagonal penalty.
47LayerDirections layer_directions(const model::Board& b, const std::vector<std::string>& dir_specs, const std::vector<std::string>& diag_specs);
