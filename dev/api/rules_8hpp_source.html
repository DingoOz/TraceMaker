15struct NetClass {
16 std::string name;
17 Coord clearance = 200'000;
18 Coord track_width = 250'000;
19 Coord via_diameter = 600'000;
20 Coord via_drill = 300'000;
21 Coord uvia_diameter = 300'000;
22 Coord uvia_drill = 100'000;
23 Coord diff_pair_width = 200'000;
24 Coord diff_pair_gap = 250'000;
25 Coord diff_pair_via_gap = 250'000;
26 bool has_diff_pair_gap = false; // set explicitly in the project (not inherited)
27 int priority = 0; // lower value = higher priority (KiCad 9+); Default has INT_MAX
28 std::string tuning_profile; // KiCad 10: the name of the class's tuning profile ("" = none)
29 bool profile_from_default = false; // tuning_profile is Default's: the class names none (KiCad's addMissingDefaults)
30 // Which of the ten parameters KiCad 10.0.6 fills from Default when unset (NET_SETTINGS::addMissingDefaults) the project sets for
31 // this class, one bit each in kNetClassFields order. A net whose classes leave one unset has Default among its effective class's
32 // constituents, which hasNetclass('Default') sees.
33 std::uint16_t fields = 0;
34 static constexpr std::uint16_t kAllFields = 0x3ff;
35 // Set by the route job for a differential tuning profile (doc 05 §58): the pair gap its pairs are routed at. Like KiCad's profile rule
36 // as KiCad 10.0.6 applies it, it relaxes the clearance between the coupled halves' tracks only, not between a track and the partner's
37 // pads (the class's diff_pair_gap still does that). 0 = none.
38 Coord tuning_pair_gap = 0;
39};

41// A KiCad 10 tuning profile (`.kicad_pro` `tuning_profiles`, doc 05 §58): per signal layer the track width and pair gap of the
42// class's nets there (0 = not set) and the propagation delay per millimetre (KiCad's time units; used only with time-domain tuning).
43struct TuningLayer {
44 std::string layer; // signal layer name, e.g. "F.Cu"
45 Coord width = 0, gap = 0;
46 std::int64_t delay = 0;
47};
48// A via's delay for one signal-layer and via-layer span, instead of the profile's delay per mm (layer names as the
49// project gives them; KiCad matches them exactly, upper layer first).
50struct TuningViaOverride {
51 std::string signal_from, signal_to, via_from, via_to;
52 std::int64_t delay = 0; // attoseconds
53};
54struct TuningProfile {
55 std::string name;
56 bool differential = false; // KiCad's PROFILE_TYPE DIFFERENTIAL (1); SINGLE (0) otherwise
57 bool time_domain = false;
58 std::vector<TuningLayer> layers;
59 std::int64_t via_delay = 0; // attoseconds per mm of via height
60 std::vector<TuningViaOverride> via_overrides;
61 const TuningViaOverride* via_override(const std::string& sf, const std::string& st, const std::string& vf, const std::string& vt) const {
62 for (const auto& o : via_overrides)
63 if (o.signal_from == sf && o.signal_to == st && o.via_from == vf && o.via_to == vt) return &o;
64 return nullptr;
65 }
66 // The width set for `layer`, 0 if none.
67 Coord width_on(const std::string& layer) const {
68 for (const auto& l : layers)
69 if (l.layer == layer) return l.width;
70 return 0;
71 }
72};

110struct DesignRules {
111 BoardMinimums minimums;
112 std::vector<NetClass> classes; // classes[0] is "Default"
113 std::vector<std::pair<std::string, std::string>> patterns; // (net-name pattern, class name), in file order
114 std::map<std::string, std::vector<std::string>> assignments; // explicit net → class names
115 std::vector<CustomRule> custom;
116 std::map<std::string, std::string> severities; // KiCad violation type -> error | warning | ignore
117 std::vector<TuningProfile> tuning_profiles; // KiCad 10 tuning profiles, as the project lists them
118 // KiCad's "Include stackup height in track length calculations" (on unless the project turns it off): vias add
119 // the stackup height between the layers they join to a net's length (drc/length.hpp, via_heights).
120 bool use_height_for_length = true;
121 // Router only, never read from a file: added to every clearance between a via and other copper. The relaxed stages of a
122 // clearance ramp lower all clearances; with this set back to the difference, vias keep the board's own clearance while
123 // tracks route thin (doc 05 §67: a thin sketch's tracks thicken, its vias parked in cells too small for them do not).
124 Coord via_clearance_extra = 0;
125 // Router only: added to the clearance between a new via and fixed copper (the board's pads, holes' copper, tracks, vias, zones), not
126 // routed copper. A relaxed stage's vias then keep the board's clearance to what legalising cannot move (doc 20 §2.6).
127 Coord via_fixed_clearance_extra = 0;
128 std::vector<std::string> warnings; // anything that could not be interpreted
129
130 const NetClass& default_class() const { return classes.front(); }
131 // Effective net class for a net name (explicit assignment, then the first matching pattern, else Default).
132 const NetClass& class_for(const std::string& net_name) const;
133 const NetClass* find_class(const std::string& name) const;
134 // Names of every class that applies to a net (explicit assignments, then matching patterns), and "Default", which
135 // KiCad 10.0.6 counts in every net's effective class: the constituents hasNetclass() searches.
136 std::vector<std::string> classes_of(const std::string& net_name) const;
137 // The tuning profile a class names, if the project has one of that name.
138 const TuningProfile* profile_of(const NetClass& c) const {
139 for (const auto& p : tuning_profiles)
140 if (!c.tuning_profile.empty() && p.name == c.tuning_profile) return &p;
141 return nullptr;
142 }
143};
