29struct BoardMinimums {

48// One constraint inside a custom rule, e.g. (constraint clearance (min 0.2mm)).
49struct Constraint {
50 std::string type; // clearance, track_width, disallow, …
51 std::optional<Coord> min, opt, max; // lengths in nm (angles/counts are stored ×1e6 as well)
52 std::vector<std::string> items; // disallow item types, or other bare words
53};

55struct CustomRule {
56 std::string name;
57 std::string condition; // raw expression text, compiled by the rule engine
58 std::string layer; // "", "outer", "inner", or a layer name
59 std::string severity;
60 std::vector<Constraint> constraints;
61};

63struct DesignRules {
64 BoardMinimums minimums;
65 std::vector<NetClass> classes; // classes[0] is "Default"
66 std::vector<std::pair<std::string, std::string>> patterns; // (net-name pattern, class name), in file order
67 std::map<std::string, std::vector<std::string>> assignments; // explicit net → class names
68 std::vector<CustomRule> custom;
69 std::map<std::string, std::string> severities; // KiCad violation type -> error | warning | ignore
70 // KiCad's "Include stackup height in track length calculations" (on unless the project turns it off): vias add
71 // the stackup height between the layers they join to a net's length (route/bus.hpp, via_heights).
72 bool use_height_for_length = true;
73 std::vector<std::string> warnings; // anything that could not be interpreted
74
75 const NetClass& default_class() const { return classes.front(); }
76 // Effective net class for a net name (explicit assignment, then the first matching pattern, else Default).
77 const NetClass& class_for(const std::string& net_name) const;
78 const NetClass* find_class(const std::string& name) const;
79};
