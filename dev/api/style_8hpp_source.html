3// Style metric (doc 05 §39.2): how tidy a set of tracks is, per connection and per board. It only measures; no
4// pass optimises it. Everything is counted in integers (nm, cross and dot products), so the numbers are the same
5// on every machine and at every thread count.
6//
7// The patterns follow what layout reviewers flag by eye; tscircuit's pcb-trace-linter (segments off the 45 degree
8// directions) and check-ugly-traces (removable jogs) were read for which patterns to count. No code of theirs is
9// used.

18struct StyleOptions {
19 // A stair-step is a run of at least `stair_min_segments` consecutive segments on one layer, each shorter than
20 // `stair_widths` track widths, whose bends turn alternately left and right.
21 int stair_widths = 4;
22 int stair_min_segments = 3;
23 // A connection is a long detour when its copper is longer than this many per cent of the shortest octilinear
24 // path between its ends.
25 int long_detour_pc = 150;
26};
27
28// One connection: copper of one net from one end to the next. An end is a pad of the net, a junction (three or
29// more track ends, or a track end on the middle of another track) or a loose end; a via with exactly two track
30// ends is passed through.
31struct ConnectionStyle {
32 model::NetId net = 0;
33 model::Point a, b; // the two ends
34 int layer_a = -1, layer_b = -1;
35 int segments = 0; // after joining collinear pieces
36 int bends = 0; // direction changes between consecutive segments on one layer
37 int acute = 0; // bends sharper than 90 degrees (the two segments enclose less than 90 degrees)
38 int stair_steps = 0; // runs, see StyleOptions
39 int short_segments = 0; // segments shorter than their own width
40 int off_angle = 0; // segments more than 0.1 degree off the eight octilinear directions
41 int arcs = 0; // arc tracks (not judged for bends or angles)
42 int vias = 0; // layer changes passed through
43 Coord length = 0; // nm, each piece rounded to the nanometre
44 Coord octile = 0; // shortest octilinear distance between the ends, nm (0 for a closed loop)
45};
46
47struct StyleReport {
48 std::vector<ConnectionStyle> connections; // sorted by net, then end points
49 int segments = 0, bends = 0, acute = 0, stair_steps = 0, short_segments = 0, off_angle = 0, arcs = 0, vias = 0;
50 int long_detours = 0;
51 Coord length = 0;
52 Coord detour_length = 0, detour_octile = 0; // summed over the connections with distinct ends
53 double bends_per_connection() const { return connections.empty() ? 0.0 : static_cast<double>(bends) / static_cast<double>(connections.size()); }
54 double detour_ratio() const { return detour_octile > 0 ? static_cast<double>(detour_length) / static_cast<double>(detour_octile) : 1.0; }
55};
