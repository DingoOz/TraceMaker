3// Repair windows (issue #56, doc 05 §75, D172): finish a nearly routed board without routing all of it again.
4//
5// A plain run keeps every track and via of the input as fixed copper, so a pin that the neighbours' fan-outs wall in
6// stays open ("fixed copper encloses the pin"). The repair lifts the unlocked copper near each open connection (whole
7// pieces: the tracks and vias of one net between its pads), routes the board again with what is left fixed and the
8// open connection first, and keeps the result only if no pads that were joined come apart and fewer connections are
9// open (rule 4). On failure the window grows. Ripping is limited to the window: everything outside stays as it was.

22struct RepairOptions {
23 std::vector<double> radii_mm{1, 2, 4}; // window radii around each end of an open connection, tried in turn
24 long step_work = 3'000'000; // work budget of each routing attempt (search expansions, rule 5)
25 int max_branches = 64; // a window that would lift more branches ends the growth for that connection
26 int max_conns = 64; // open connections tried at most (the first ones of the router's list)
27 double time_limit_s = 0; // > 0: no attempt starts after this many seconds (safety only; work decides)
28 std::vector<model::NetId> keep_nets; // nets whose copper is never lifted (coupled pairs, bus members)
29};

31// One routing attempt: an open connection, a window radius, what it lifted, and whether it was kept.
32struct RepairStep {
33 std::string conn; // "NET A-B"
34 double radius_mm = 0;
35 int pieces = 0, tracks = 0, vias = 0; // copper lifted (pieces: branches)
36 int open_before = 0, open_after = 0; // open connections of the board (pads not joined), before and after the attempt
37 bool split = false; // some pads that were joined came apart
38 bool kept = false;
39 long work = 0;
40};

42struct RepairResult {
43 bool changed = false;
44 // The board's own tracks and vias that were lifted (indices into the `base` board given), ascending.
45 std::vector<int> ripped_tracks, ripped_vias;
46 // The new copper after the repair: what is left of the run's new copper, then the repair's, in commit order.
47 std::vector<model::Track> tracks;
48 std::vector<model::Via> vias;
49 int open_before = 0, open_after = 0;
50 // Of the last kept attempt (pad indices of `base`): the router's open connections and its explanations.
51 std::vector<route::RouteResult::Unrouted> unrouted;
52 std::vector<std::string> failures;
53 std::vector<RepairStep> steps;
54 long work = 0;
55 double seconds = 0;
56 nlohmann::json summary() const;
57};
