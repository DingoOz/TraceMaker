3// Replay log (design doc 09 §2, decision D45): the viewer protocol's JSON messages, one per line, each with a
4// leading "t" (seconds since the recording started). Written plain (`.jsonl`) or as a sequence of independent
5// zstd frames (`.zst`, about 10-20x smaller). Each frame holds whole lines and the board snapshot (the keyframe)
6// is a frame of its own, so a run that is killed loses at most the open frame, and any standard tool reads the
7// log (`zstd -dc run.jsonl.zst | jq`).
8#include <cstdint>
9#include <cstdio>
10#include <string>
11#include <string_view>

13// zstd contexts (zstd.h stays out of this header).
14struct ZSTD_CCtx_s;
15struct ZSTD_DCtx_s;
