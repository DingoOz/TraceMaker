3// Routability estimate for placement (design doc 04 §3 B `R(x)`, E `β·RUDY overflow`, G).
4//
5// RUDY (Rectangular Uniform wire DensitY; Spindler & Johannes, "Fast and accurate routing demand estimation
6// for efficient routability-driven placement", DATE 2007): each signal net's half-perimeter wirelength is spread
7// uniformly over its pin bounding box, so the demand of a bin is Σ_nets HPWL · area(box ∩ bin) / area(box).
8// Every pin also consumes a fixed amount of track length in its bin (pin density). Capacity per bin is the
9// track length the copper layers offer there (layers × bin² / pitch × usable fraction × fraction inside the
10// outline). The cost is the total overflow Σ max(0, demand − capacity), all in integer nm so incremental and
11// from-scratch evaluation agree exactly (CLAUDE.md rules 2 and 3).
12//
13// Power nets (PNet::signal false) are left out (they are expected on pours or planes; the same reason they get a low HPWL
14// weight). The router-in-the-loop (G) lowers the capacity of bins where the router failed (`scale_bins`), which
15// is the PCB analogue of RePlAce's cell inflation in overflowed tiles.

64namespace detail {
65// RUDY box: the pin box widened to at least one bin per axis, so aligned pins do not give infinite density.
66inline Box rudy_box(const CongestionMap& m, const Box& b) {
67 Box r = b;
68 if (r.x1 - r.x0 < m.cell) {
69 const Coord c = (r.x0 + r.x1) / 2;
70 r.x0 = c - m.cell / 2;
71 r.x1 = r.x0 + m.cell;
72 }
73 if (r.y1 - r.y0 < m.cell) {
74 const Coord c = (r.y0 + r.y1) / 2;
75 r.y0 = c - m.cell / 2;
76 r.y1 = r.y0 + m.cell;
77 }
78 return r;
79}
80} // namespace detail

82template <class Touch>
83void add_box_demand(const CongestionMap& m, const Box& b, int sign, std::vector<std::int64_t>& demand, Touch&& touch) {
84 if (b.empty()) return;
85 const Coord hp = (b.x1 - b.x0) + (b.y1 - b.y0);
86 if (hp <= 0) return;
87 const Box r = detail::rudy_box(m, b);
88 const geom::i128 area = static_cast<geom::i128>(r.x1 - r.x0) * (r.y1 - r.y0);
89 auto cx = [&](Coord v) { return static_cast<int>(std::clamp<Coord>((v - m.ox) / m.cell, 0, m.nx - 1)); };
90 auto cy = [&](Coord v) { return static_cast<int>(std::clamp<Coord>((v - m.oy) / m.cell, 0, m.ny - 1)); };
91 const int x0 = cx(r.x0), x1 = cx(r.x1 - 1), y0 = cy(r.y0), y1 = cy(r.y1 - 1);
92 for (int y = y0; y <= y1; ++y) {
93 // Bins at the grid border absorb everything beyond it.
94 const Coord by0 = y == 0 ? r.y0 : std::max(r.y0, m.oy + y * m.cell);
95 const Coord by1 = y == m.ny - 1 ? r.y1 : std::min(r.y1, m.oy + (y + 1) * m.cell);
96 if (by1 <= by0) continue;
97 for (int x = x0; x <= x1; ++x) {
98 const Coord bx0 = x == 0 ? r.x0 : std::max(r.x0, m.ox + x * m.cell);
99 const Coord bx1 = x == m.nx - 1 ? r.x1 : std::min(r.x1, m.ox + (x + 1) * m.cell);
100 if (bx1 <= bx0) continue;
101 const geom::i128 ov = static_cast<geom::i128>(bx1 - bx0) * (by1 - by0);
102 const auto d = static_cast<std::int64_t>(static_cast<geom::i128>(hp) * ov / area);
103 if (d == 0) continue;
104 const auto k = static_cast<std::size_t>(y * m.nx + x);
105 touch(k);
106 demand[k] += sign * d;
107 }
108 }
109}
