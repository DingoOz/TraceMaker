3// Exact integer geometry for clearance checks (design doc 03 §1).
4//
5// Every copper shape is a "rounded core": a point, an open polyline or a closed polygon, inflated by a
6// radius r. Tracks are 2-point polylines with r = width/2, round pads are points, ovals are segments,
7// round-rect pads are rectangles shrunk by the corner radius and inflated back. Gap tests compare squared
8// distances in 128-bit integers, so "closer than" decisions are exact.
9#include <cstdint>
10#include <vector>
11
12#include "geom/point.hpp"

18struct Shape {
19 std::vector<Point> pts; // 1 point, open polyline (>= 2), or closed polygon (>= 3)
20 Coord r = 0; // inflation radius
21 bool closed = false; // polygon interior is part of the shape
22 Box box; // bounding box including r

24 static Shape point(Point p, Coord r);

29 // Reset to a disk, reusing point storage for scratch legality checks.

31};

33// Orientation of c relative to a→b: >0 left (counter-clockwise in y-up math terms), <0 right, 0 collinear.
34inline i128 orient(Point a, Point b, Point c) {
35 return static_cast<i128>(b.x - a.x) * (c.y - a.y) - static_cast<i128>(b.y - a.y) * (c.x - a.x);
36}
