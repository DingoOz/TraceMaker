29// Bus patterns select nets by their full name (KiCad's, with the sheet path: "/D0", "/MEM/DQ3").
30// [N-M] or [N..M], N <= M decimal numbers: an index range, matching any decimal number in it ("D[0-15]" matches
31// D0 .. D15, and D07 too). A range needs both bounds; "[[0-7]]" is a range in literal brackets,
32// for nets named A[0] .. A[7].
33// * and ?: as in KiCad's net-class patterns (any run of characters, any one character).
34// A pattern without an index range is matched exactly as a KiCad net-class pattern (model::pattern_match: wildcards,
35// or a full regular expression). In a pattern with a range everything outside the range and the wildcards is literal.
36bool bus_pattern_match(const std::string& pattern, const std::string& net_name);

38// "NAME=PATTERN" as given to --bus. Throws std::runtime_error when there is no '=', the name or pattern is empty, or
39// fewer than two nets with pads match.
40BusGroup bus_from_option(const model::Board& b, const std::string& spec);

42// Automatic detection (--buses), conservative: nets whose names differ only in a trailing decimal index ("D0", "A[3]",
43// "/MEM/DQ12"), at least `min_members` of them with pads on the same two components. KiCad's generated names
44// ("Net-(U1-Pad3)", "unconnected-...") are never grouped. One stem may give several groups (D0-7 to one memory,
45// D8-15 to another); they are then named STEM(REF-REF). Groups come in natural order of their names.
46std::vector<BusGroup> detect_buses(const model::Board& b, int min_members = 4);

49// (the smallest maximum wins); rules restricted to differential pairs (`within_diff_pairs`) do not count.
50std::optional<Coord> bus_rule_skew(const drc::RuleEngine& re, const BusGroup& g);

52// The length a via adds to its net in KiCad's length calculation, which its `skew` check uses (KiCad 10,
53// BOARD_STACKUP::GetLayerDistance): the stackup between the two copper layers the via joins, with the copper of an
54// outer layer in full and of an inner layer by half. A board without a stackup block gets KiCad's default one
55// (35 um copper, 10 um per solder mask, the rest of the board thickness shared evenly between the dielectrics).
56// Empty (every via 0) when the project turns "use height for length calculations" off.
57struct ViaHeights {
58 std::vector<Coord> from, to; // per copper layer: where a via starting / ending on it begins / ends, nm from the top
59 Coord between(int a, int b) const; // copper indices, either order; 0 for a == b
60};
61ViaHeights via_heights(const model::Board& b, const model::DesignRules& rules);
62// The via length of one net on the given copper, as KiCad sums it: each via counts between the outermost layers on
63// which a track of the net ends inside the via's pad, or a pad of the net touches the via (a surface pad's own
64// layer, its footprint's side for a through-hole pad); a via that joins nothing on a second layer adds nothing.
65double net_via_length(const model::Board& b, const ViaHeights& vh, model::NetId net, const std::vector<model::Track>& tracks,
66 const std::vector<model::ArcTrack>& arcs, const std::vector<model::Via>& vias);
67
68// A group as routed, measured on copper. Length is what KiCad's skew check compares: track length (arcs included)
69// plus the via length above, so a member with more vias than the others is longer by their height. KiCad also
70// adds pad-to-die lengths and shortens a track end inside a pad or via; neither is modelled (micrometres, or zero).
71struct BusMember {
72 model::NetId net = 0;
73 bool routed = false; // every pad of the net is joined
74 double length = 0; // nm, tracks and vias
75 double via_length = 0; // nm, the vias' part of `length`
76 int vias = 0;
77 model::LayerMask layers = 0; // copper layers carrying tracks of the net
78 bool followed = false; // routed with the group's reference via count and layers
79};
80struct BusStats {
81 std::vector<BusMember> members; // group order
82 int routed = 0;
83 // The reference is the via count and layer set most members share (ties: the first member's in group order).
84 // `followed` members have exactly that; `left` are routed otherwise ("left the bus").
85 int followed = 0, left = 0;
86 int ref_vias = 0;
87 model::LayerMask ref_layers = 0;
88 int vias_min = 0, vias_max = 0; // over routed members
89 double length_min = 0, length_max = 0;
90 double skew() const { return length_max - length_min; } // over routed members
91};

93BusStats measure_bus(const model::Board& b, const ViaHeights& vh, const std::vector<model::Track>& tracks, const std::vector<model::ArcTrack>& arcs, const std::vector<model::Via>& vias,
94 const std::vector<std::uint8_t>& net_routed, const BusGroup& g);
