16struct CourtyardRegion {
17 // The courtyard is the union of the parts. The rings of one part use even/odd filling (a ring inside another
18 // is a hole); outlines that overlap or touch are parts of their own, as KiCad merges them.
19 std::vector<std::vector<std::vector<geom::Point>>> parts;
20 geom::Box box;
21 // The side has courtyard graphics, but they are not closed, simple outlines (or use a Bezier curve): no
22 // parts, and the rule engine warns that courtyard conditions cannot match this footprint.
23 bool unreadable = false;
24};
25struct CourtyardEntry {
26 std::string reference, lib_id;
27 bool back = false; // footprint flipped to the bottom side
28 std::array<CourtyardRegion, 2> sides; // physical F.CrtYd, B.CrtYd
29};
