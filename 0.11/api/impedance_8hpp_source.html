3// Closed-form transmission-line impedance for PCB traces (design doc 15 §5.3) and the IPC-2221 width for
4// current (§5.4). Quasi-static (no dispersion: the targets are 45-100 ohm lines below a few GHz), lossless,
5// no solder mask (it lowers outer-layer impedance by roughly 1-3 ohm; KiCad's calculator also leaves it off
6// by default). Geometry in integer nanometres; impedances in ohm.
7//
8// Models (each cited at its definition in impedance.cpp):
9// microstrip Hammerstad & Jensen 1980, with their finite-thickness correction
10// stripline Cohn 1954 (wide strip, thick) / Wheeler 1978 (narrow strip), asymmetric by the image
11// split of Wadell 1991 §3.5.3 (two virtual centred striplines in parallel)
12// coupled microstrip Kirschning & Jansen 1984 static even/odd mode, Jansen 1978 thickness correction,
13// Bahl & Garg 1977 thickness filling-factor term
14// coupled stripline Cohn 1955 zero-thickness even/odd, Cohn 1960 thickness correction, asymmetric by the
15// same image split
16// grounded coplanar Ghione & Naldi 1983/1984 conformal mapping, Gupta et al. 1996 thickness correction
17// Solving for a width or gap is a bisection on integer nanometres: deterministic and exact to 1 nm.

33struct Pair { // symmetric edge-coupled pair
34 double z_even = 0, z_odd = 0;

36 double zdiff() const { return 2.0 * z_odd; } // differential = 2 x odd mode
37};

39// Microstrip: strip of width w and thickness t on a dielectric of height h (strip bottom to plane), εr er.
40Line microstrip(Coord w, Coord h, Coord t, double er);
41// Stripline between two planes: h1 dielectric from the upper plane to the strip, h2 from the strip to the lower
42// plane (the plane spacing is h1 + t + h2); symmetric when h1 == h2. Homogeneous dielectric: eps_eff = er.
43Line stripline(Coord w, Coord h1, Coord h2, Coord t, double er);
44// Edge-coupled microstrip: two strips of width w, edge-to-edge gap s.
45Pair coupled_microstrip(Coord w, Coord s, Coord h, Coord t, double er);
46// Edge-coupled stripline (broadside layout not modelled).
47Pair coupled_stripline(Coord w, Coord s, Coord h1, Coord h2, Coord t, double er);
48// Grounded (conductor-backed) coplanar waveguide: strip w, gap g to the coplanar ground on both sides, plane at h.
49Line gcpw(Coord w, Coord g, Coord h, Coord t, double er);
