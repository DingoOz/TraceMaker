3// The user override file (design doc 15 §3.5 level 1, §6.3): the user disables rules, forces or forbids a
4// category on a part, or changes a rule parameter. The engine reads JSON (it has no YAML parser, like the
5// catalogue); the YAML form of doc 15 §6.3 maps 1:1 to it and scripts/crules_override.py converts it.
6//
7// {
8// "version": 1,
9// "disable": ["USB2-04@J2", "XTAL-04", "mounting_hole", "crystal@Y2"],
10// "assert": {"J5": "usb2"}, // or [{"category": "usb2", "ref": "J5"}]
11// "deny": {"U7": "buck"}, // or [{"category": "buck", "ref": "U7"}]
12// "set": [{"rule": "XTAL-02", "param": "max_mm", "value": 5},
13// {"rule": "XTAL-01@Y1", "param": "max_mm", "value": 8}]
14// }
15//
16// Everything is validated against the catalogue when the file is read: unknown keys, rule ids, categories,
17// parameters and value types are errors, never ignored (a typo must not silently drop an override). References
18// are validated against the board by check_override_refs (detect() calls it).

31struct OverrideEntry {
32 enum class Kind : std::uint8_t { Disable, Assert, Deny, Set };
33 Kind kind = Kind::Disable;
34 std::string rule; // Disable (one rule) and Set: the rule id
35 std::string category; // the category id (for rule entries: the rule's category)
36 std::string ref; // anchor reference; empty = every instance (Disable, Set)
37 std::string param; // Set: parameter name
38 nlohmann::ordered_json value; // Set: new value
39 std::string text; // the entry as the report shows it, e.g. "disable XTAL-04@Y1"
40};
41
42struct Overrides {
43 std::string source; // file name (messages and report)
44 std::vector<OverrideEntry> entries; // file order: disable, assert, deny, set
45 bool empty() const { return entries.empty(); }
46};

51// Reads a JSON file; a .yaml/.yml file is refused with the conversion command.

53// Every reference must name a footprint of the board (throws otherwise, listing the unknown ones).
